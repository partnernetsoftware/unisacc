#!/bin/sh
set -eu
cd /mnt
export SEED_DIR=/mnt/out/f3-model STRICT=1
candidate=/mnt/out/f3-model/comb-build/stage3
stage2=/mnt/out/f3-model/comb-build/stage2
cmp out/f3-model/unisacc-seed.com "$stage2/unisacc-next.com"
cmp "$stage2/unisacc-next.com" "$candidate/unisacc-next.com"
cmp "$stage2/unisacc-next.com.build.json" "$candidate/unisacc-next.com.build.json"
cmp out/f3-model/stage3/unisacc.com "$candidate/unisacc-next.com"
cmp out/f3-model/stage3/unisacc.com.build.json "$candidate/unisacc-next.com.build.json"
sh exec/c/comboot.sh shard fixedpoint
sh release/tools/installpair.sh "$candidate" /mnt/out/f3-model /mnt
cmp unisacc.com "$candidate/unisacc-next.com"
cmp unisacc.com.build.json "$candidate/unisacc-next.com.build.json"
out/f3-model/seedbin/ident /mnt check /mnt/unisacc.com
out/f3-model/seedbin/ident /mnt check "$candidate/unisacc-next.com"
sha256sum out/f3-model/unisacc-seed.com unisacc.com out/f3-model/stage3/unisacc.com
sh exec/c/comboot.sh report
