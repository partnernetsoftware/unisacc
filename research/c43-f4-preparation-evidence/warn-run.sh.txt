#!/bin/bash
set -euo pipefail
cd /mnt
W=/mnt/out/warn-recovery-20261024c
B=/mnt/tests/bound
T=/mnt/out/warn-recovery-20261024a/seed-tools
P=/opt/unisacc-reference/bin/pypy3
[ -z "${UNISA_JOBS+x}" ]
[ "$(git rev-parse HEAD)" = "$(git rev-parse 62aa49e5a3)" ]
git diff --quiet
sha256sum --quiet -c out/final-accept-inputs.sha256
sha256sum "$T/gen" "$T/tbl" "$T/net" "$P" > "$W/tool-inputs.sha256"
"$P" --version > "$W/reference-version.txt" 2>&1
step() {
 local label=$1; shift
 [ ! -e "$W/$label.log" ] || { echo "refuse existing step $label"; exit 2; }
 printf 'STEP_BEGIN %s\n' "$label"
 set +e
 "$B" --status "$W/$label.status" 55 "$@" > "$W/$label.log" 2>&1
 local rc=$?
 set -e
 printf '%s\t%s\n' "$label" "$rc" >> "$W/steps.tsv"
 printf 'STEP_END %s rc=%s\n' "$label" "$rc"
 if [ "$rc" -ne 0 ]; then tail -8 "$W/$label.log"; exit "$rc"; fi
 sha256sum --quiet -c out/final-accept-inputs.sha256
 sha256sum --quiet -c "$W/tool-inputs.sha256"
}
for route in warnparse errorparse; do
 if [ "$route" = warnparse ]; then flags=(--warnings --errors); group=1; else flags=(--errors); group=2; fi
 if [ "$route" = warnparse ]; then
  CJSON=/mnt/out/warn-recovery-20261024a/warnparse.c.json
  PJSON=$W/warnparse.pypy.json
  [ "$(cat "$W/warnparse-pypy.status")" = 0 ]
 else
  CJSON=$W/$route.c.json; PJSON=$W/$route.p.json
  step "$route-c-json" "$T/gen" parse2 "$CJSON" "${flags[@]}"
  step "$route-p-json" "$P" exec/build/gen.py parse2 "$PJSON" "${flags[@]}"
 fi
 cmp "$CJSON" "$PJSON"
 step "$route-c-tbl" "$T/tbl" "$CJSON" "$W/$route.c.tbl"
 step "$route-p-tbl" "$P" exec/c/tbl.py "$PJSON" "$W/$route.p.tbl"
 cmp "$W/$route.c.tbl" "$W/$route.p.tbl"
 step "$route-c-net" "$T/net" "$W/$route.c.tbl" "$W/$route.c.net"
 step "$route-p-net" "$P" exec/c/net.py "$W/$route.p.tbl" "$W/$route.p.net"
 cmp "$W/$route.c.net" "$W/$route.p.net"
 sha256sum "$CJSON" "$PJSON" "$W/$route.c.tbl" "$W/$route.p.tbl" "$W/$route.c.net" "$W/$route.p.net" > "$W/$route-parity.sha256"
 printf 'PARITY_OK %s json table network; no old failure overwritten\n' "$route"
 step "$route-cache" python3 exec/c/compilerpack.py --prepare-only --part "$group/5" --seed-tools "$T" --keep-json --o1 out/f3-model/shared/o1.net --include include --shared-e2 out/f3-model/shared/e2.net --shared-nativeabi out/f3-model/shared/nativeabi.net -o /dev/null out/f3-model/lnx-x86_64/route.tsv
 step "$route-hit" "$B" 41 python3 exec/parse2/warnprep.py "$group"
done
step locations-original python3 exec/parse2/warnprep.py 3
step locations-hit "$B" 41 python3 exec/parse2/warnprep.py 3
printf 'PREPARATION_RECOVERY_COMPLETE; original F4 failures remain; consumer assertions not yet claimed\n'
