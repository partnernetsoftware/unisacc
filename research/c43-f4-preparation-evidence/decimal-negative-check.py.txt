import os,pathlib,subprocess,hashlib,json
root=pathlib.Path('/mnt'); out=root/'out'
hlog=(out/'prepare-harness.log').read_text().strip().splitlines()[-1]
cache=pathlib.Path(hlog.split(' ',2)[2]); leaf=cache/'h.net'; backup=cache/'h.net.negative-backup'
assert not backup.is_file(); before=hashlib.sha256(leaf.read_bytes()).hexdigest(); rows=[]
def run(label,env=None):
 p=subprocess.run(['tests/bound','41','out/decimal-v2','--require-cached'],env=env,capture_output=True)
 (out/(label+'.log')).write_bytes(p.stdout+p.stderr)
 (out/(label+'.status')).write_text(str(p.returncode)+'\n')
 assert p.returncode==75,(label,p.returncode,p.stdout,p.stderr)
 assert b'NEED_PREPARE decimal:' in p.stderr,(label,p.stderr)
 rows.append({'label':label,'rc':p.returncode})
e=dict(os.environ);e['CFLAGS']='-DDECIMAL_IDENTITY_NEGATIVE=1';run('negative-identity',e)
leaf.rename(backup)
try:
 run('negative-missing')
 data=bytearray(backup.read_bytes());data[len(data)//2]^=1;leaf.write_bytes(data)
 run('negative-corrupt');leaf.unlink()
 leaf.symlink_to(backup.name);run('negative-symlink');leaf.unlink()
finally:
 try:leaf.lstat()
 except FileNotFoundError:pass
 else:leaf.unlink()
 backup.rename(leaf)
assert hashlib.sha256(leaf.read_bytes()).hexdigest()==before
(out/'negative-results.json').write_text(json.dumps({'original_restored_sha256':before,'checks':rows},indent=2)+'\n')
print('NEGATIVE_CHECKS',len(rows),'all NEED_PREPARE75; original restored',before)
