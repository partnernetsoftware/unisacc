#!/bin/sh
set -eu
cd /mnt
step() {
 label=$1
 limit=$2
 shift 2
 printf 'BEGIN %s\n' "$label"
 set +e
 tests/bound "$limit" "$@" > "out/$label.log" 2>&1
 rc=$?
 set -e
 printf '%s\n' "$rc" > "out/$label.status"
 printf 'END %s rc=%s\n' "$label" "$rc"
 tail -6 "out/$label.log"
 [ "$rc" -eq 0 ] || exit "$rc"
}
step compile 55 sh unisacc.com -o out/decimal-v2 tests/decimalcheck.cx
step prepare-tools 55 out/decimal-v2 --prepare --part tools
step prepare-harness 55 out/decimal-v2 --prepare --part harness
step consume 41 out/decimal-v2 --require-cached --evidence /mnt/out/c-evidence
