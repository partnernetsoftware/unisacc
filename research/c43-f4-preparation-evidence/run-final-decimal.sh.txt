#!/bin/sh
set -eu
cd /mnt
step() {
 label=$1; limit=$2; shift 2
 printf 'BEGIN %s\n' "$label"
 set +e
 tests/bound "$limit" "$@" > "out/final-$label.log" 2>&1
 rc=$?
 set -e
 printf '%s\n' "$rc" > "out/final-$label.status"
 printf 'END %s rc=%s\n' "$label" "$rc"
 tail -5 "out/final-$label.log"
 [ "$rc" -eq 0 ] || exit "$rc"
}
step decimal-compile 55 sh unisacc.com -o out/decimal-final tests/decimalcheck.cx
step decimal-tools 55 out/decimal-final --prepare --part tools
step decimal-harness 55 out/decimal-final --prepare --part harness
step decimal-consumer 41 out/decimal-final --require-cached
