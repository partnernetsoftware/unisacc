#pragma unisa use text.text
int main(int c, char **v) {
    int n = 0;
    if (tx_eq(v[0], "a", 1L << 20)) n++;     /* yes */
    if (!tx_eq(v[0], "b",  1L << 20 )) n++;  /* yes */
    if (tx_eq(v[0], "c", 8)) n++;            /* no: cap 8 */
    if (tx_eq(v[0], "d", 1L<<20)) n++;       /* no: other spelling */
    return n;
}
