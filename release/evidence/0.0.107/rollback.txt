# Rollback plan for unisacc v0.0.107 (chain_rev=63354b4a2cb707ecc923a89f0df6030149fa697f, ape_sha256=b773c7e221ab059e935276f41c0ac4070737f2ae6c548cd56ea8b6b0acc53c63)
# 状态: Step 1 X 前写妥并核验; 涵盖未公开前(草稿/标签/X提交)与公开后(切回草稿/Latest/Y2-Y1-X前向撤销/本机根回退)两套撤回办法。

(0) 未公开前(Step 1 X / Step 2 draft / Step 3 CI 阶段)撤回办法:
    # a. 若 CI 仍在运行先取消:
    gh -R partnernetsoftware/unisacc run cancel <CI_RUN_ID>
    # b. 保持草稿 release v0.0.107 为 draft 私有(或需清理草稿与远端未公开 tag 时执行 gh -R partnernetsoftware/unisacc release delete v0.0.107 --yes --cleanup-tag; 按 release-exec 缺省合同优先保持 draft=true 不公开):
    gh -R partnernetsoftware/unisacc release edit v0.0.107 --draft=true
    # c. 在 v0.0.107 分支(及已同步分支)前向 revert X 提交(禁止 reset/stash/checkout/force-push):
    git revert --no-edit <X_COMMIT_SHA>
    DEVSVC_OWNER=cc sh ./unisacc.com tools/devsvc.cx -- git push HEAD

(1) 公开后(Step 5 publish 之后)将 GitHub Release v0.0.107 转回草稿并恢复 Latest 指向 v0.0.106:
    gh -R partnernetsoftware/unisacc release edit v0.0.107 --draft=true
    gh -R partnernetsoftware/unisacc release edit v0.0.106 --draft=false --latest

(2) 读回核验 v0.0.106 为 Latest 且 v0.0.107 为 Draft:
    gh -R partnernetsoftware/unisacc release view v0.0.107 --json tagName,isDraft,isLatest,assets
    gh -R partnernetsoftware/unisacc release view v0.0.106 --json tagName,isDraft,isLatest,assets
    gh api repos/partnernetsoftware/unisacc/releases/latest --jq .tag_name

(3) 共享 main 与车树上从新到旧前向撤销发布提交(Y2、Y1、X, 禁止 force-push):
    git revert --no-edit <Y2_COMMIT_SHA>
    git revert --no-edit <Y1_COMMIT_SHA>
    git revert --no-edit <X_COMMIT_SHA>
    DEVSVC_OWNER=cc sh ./unisacc.com tools/devsvc.cx -- git push HEAD

(4) 本机根回退(若已将 v0.0.107 APE 安装至本机检出根):
    DEVSVC_OWNER=cc sh ./unisacc.com tools/install-ape.cx -- . --rollback --write
