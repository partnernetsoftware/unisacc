2. 统一低负载门禁包装模板 (`run_under_load_cap`, 覆盖 Phase 2b/2c/4a/4b/5a/6, B7):
   ```sh
   run_under_load_cap() {
       cap="$1"; shift
       l1_before="$(cut -d' ' -f1 /proc/loadavg)"
       heavy="$(pgrep -fa 'candidate-chain|release-exec|source-job-wave|pc-wave|tools-compile' || true)"
       printf 'LOAD_GATE_BEFORE\tcap=%s\tload1=%s\theavy=%s\n' "$cap" "$l1_before" "${heavy:-none}"
       if [ -n "$heavy" ] || awk "BEGIN{exit !($l1_before >= $cap)}"; then
           printf 'LOAD_GATE_REFUSE\tphase=before\tcap=%s\tload1=%s\n' "$cap" "$l1_before" >&2
           return 2
       fi
       "$@"
       rc=$?
       l1_after="$(cut -d' ' -f1 /proc/loadavg)"
       printf 'LOAD_GATE_AFTER\tcap=%s\tload1=%s\trc=%d\n' "$cap" "$l1_after" "$rc"
       if awk "BEGIN{exit !($l1_after >= $cap)}"; then
           printf 'LOAD_GATE_REFUSE\tphase=after\tcap=%s\tload1=%s\n' "$cap" "$l1_after" >&2
           return 2
       fi
       return "$rc"
   }
   ```
   - 凡标明 `load1 < 8.0` 的命令一律以 `run_under_load_cap 8.0 <cmd...>` 运行; `LOAD_GATE_REFUSE` 触发即标 `UNMEASURED` 并等待空载, 严禁采信超载结果。
